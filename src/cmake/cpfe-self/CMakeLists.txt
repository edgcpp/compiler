# Self-compilation of the front end.  cpfe (whose back end generates C)
# translates the front end's own sources to C, and the C compiler builds that C
# into cpfe-self.  This is the path a user takes to build C++ code for a system
# that has only a C compiler, so it checks that the front end can still be
# built that way.
#
# The cpfe-self-check target then has cpfe-self translate the same sources and
# compares the result with the C generated by cpfe.  cpfe and cpfe-self are
# built from the same sources, so they must generate the same C, apart from
# the differences that are expected between any two runs (see
# edg-compare-gen-c).

# The front end options are assembled from directory properties, which may
# contain per-configuration generator expressions in a multi-configuration
# build.
if(CMAKE_CONFIGURATION_TYPES)
  message(FATAL_ERROR
          "EDG_SELF_COMPILE requires a single-configuration generator (e.g., "
          "Ninja).")
endif()
# With the daemon, cpfe is only the client.
if($CACHE{MULTI_THREADED_DAEMON})
  message(FATAL_ERROR
          "EDG_SELF_COMPILE is not supported with MULTI_THREADED_DAEMON.")
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)

# The generated C is compiled with the C compiler, so the front end emulates
# that compiler (which also allows it to process the compiler's headers).
if(MSVC)
  # The front end must lay out structs as MSVC does, but the debug macro
  # configurations turn off Microsoft bit-field allocation.
  configure_definitions_as_values("base" "$CACHE{EDG_MACRO_CONF}")
  if("TARG_MICROSOFT_BIT_FIELD_ALLOCATION=0" IN_LIST
     configured_cmakedef_definitions)
    message(FATAL_ERROR
            "EDG_SELF_COMPILE with MSVC requires a macro configuration that "
            "uses Microsoft bit-field allocation (e.g., that of the "
            "windows-msvc-release preset).")
  endif()
  # As the C++ build does (see /wd4996), do not warn about the C library
  # routines that Microsoft deprecates, or about the "#pragma warning"
  # directives intended for MSVC.
  set(dialect_options
      --microsoft --microsoft_version=${MSVC_VERSION}
      -D_CRT_SECURE_NO_WARNINGS -D_CRT_NONSTDC_NO_WARNINGS
      --diag_suppress=unrecognized_pragma)
  # The C++/CLI support (ms_metadata.cpp) is implemented with the Microsoft
  # C++ library, ATL, and COM, which have no C counterparts.  Since it is not
  # needed on systems without a C++ compiler, leave it out (along with the
  # C++/CX support, which depends on it) by overriding the macro
  # configuration with the settings from the cppcli/off feature.
  set(macro_conf_root_path "${CMAKE_SOURCE_DIR}/cmake/macro-conf")
  set(cppcli_off_path
      "${macro_conf_root_path}/support/feature/cppcli/off.cmakedef")
  execute_process(COMMAND "${TOOL_PATH_EDG_CMAKEDEF}"
                          "${macro_conf_root_path}"
                          "${cppcli_off_path}"
                          print-values
                  OUTPUT_VARIABLE front_end_definitions
                  OUTPUT_STRIP_TRAILING_WHITESPACE
                  ERROR_VARIABLE cmakedef_messages
                  RESULTS_VARIABLE cmd_status_code)
  if(NOT cmd_status_code EQUAL "0")
    message(FATAL_ERROR "Failed to process cmakedef:\n${cmakedef_messages}")
  endif()
  string(REPLACE "\n" ";" front_end_definitions "${front_end_definitions}")
  list(TRANSFORM front_end_definitions PREPEND "-D")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
               "${cppcli_off_path}")
  # The C compiler's include directories are in the INCLUDE environment
  # variable (set by the Visual Studio developer environment).
  file(TO_CMAKE_PATH "$ENV{INCLUDE}" host_include_dirs)
elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR
       CMAKE_C_COMPILER_ID STREQUAL "Clang")
  # Encode the C compiler version in the form the front end expects (e.g.,
  # 15.3.1 becomes 150301).
  if(NOT CMAKE_C_COMPILER_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)")
    message(FATAL_ERROR
            "Unrecognized C compiler version "
            "\"${CMAKE_C_COMPILER_VERSION}\".")
  endif()
  math(EXPR compiler_version
       "${CMAKE_MATCH_1} * 10000 + ${CMAKE_MATCH_2} * 100 + ${CMAKE_MATCH_3}")
  if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    set(dialect_options --g++ --gnu_version=${compiler_version})
  else()
    set(dialect_options --clang --clang_version=${compiler_version})
  endif()
  set(front_end_definitions "")
  set(host_include_dirs ${CMAKE_C_IMPLICIT_INCLUDE_DIRECTORIES})
else()
  message(FATAL_ERROR
          "EDG_SELF_COMPILE does not support the C compiler "
          "\"${CMAKE_C_COMPILER_ID}\".")
endif()
list(REMOVE_ITEM host_include_dirs "")

# Options used for every translation.  Each translation unit gets the template
# instantiations it uses, so no prelinker is needed (see the handling of the
# duplicates below).  The EDG C++ headers come first so that, e.g., <new> is
# EDG's.
set(common_options
    ${dialect_options}
    --no_exceptions --no_rtti
    --instantiate=used
    "--sys_include=${CMAKE_SOURCE_DIR}/include_c++")
list(TRANSFORM host_include_dirs PREPEND "--sys_include="
     OUTPUT_VARIABLE host_include_options)
list(APPEND common_options ${host_include_options})
if(NOT "$CACHE{EDG_BASE}" STREQUAL "")
  # The EDG base supplies the C compiler's predefined macros.
  list(APPEND common_options "--edg_base_dir=$CACHE{EDG_BASE}")
endif()

# The front end sources are translated with the same macros and include
# directories as when they are compiled into cpfe.
get_directory_property(cpfe_definitions
                       DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../cpfe
                       COMPILE_DEFINITIONS)
list(TRANSFORM cpfe_definitions PREPEND "-D")
get_target_property(cpfe_include_dirs cpfe INCLUDE_DIRECTORIES)
list(TRANSFORM cpfe_include_dirs PREPEND "-I")
set(front_end_options
    --c++${CMAKE_CXX_STANDARD}
    ${common_options}
    ${cpfe_definitions} ${front_end_definitions}
    ${cpfe_include_dirs})

# The generated code calls routines from the EDG runtime library (e.g., the
# placement operator new and, with the ABI used on Windows, the allocation and
# static initialization routines).  The runtime library is not built on every
# host, so translate the sources of those routines as well (cpfe-self links
# with only the ones it needs).  runtime_vars.c stands in for lib_src/vars.c,
# which cannot be translated in Microsoft mode.
set(runtime_source_names
    delete dtor_list error main memzero munch_ctors munch_dtors new
    newhandler placenew sized_del static_init)
list(TRANSFORM runtime_source_names
     REPLACE "(.+)" "${CMAKE_SOURCE_DIR}/lib_src/\\1.c"
     OUTPUT_VARIABLE runtime_source_files)
list(APPEND runtime_source_files "${CMAKE_CURRENT_SOURCE_DIR}/runtime_vars.c")
# Exception handling is disabled in the runtime as it is in the front end.
set(runtime_options
    --c++23 --building_runtime
    ${common_options}
    -DEXCEPTION_HANDLING=0
    "-I${CMAKE_SOURCE_DIR}/lib_src")

# The C generated by cpfe (and compiled into cpfe-self) and by cpfe-self (for
# the comparison).
set(cpfe_c_dir "${CMAKE_CURRENT_BINARY_DIR}/cpfe-c")
set(cpfe_self_c_dir "${CMAKE_CURRENT_BINARY_DIR}/cpfe-self-c")
file(MAKE_DIRECTORY "${cpfe_c_dir}" "${cpfe_self_c_dir}")

function(add_translation front_end_target source_file output_file)
  # Add a command that translates source_file to C (in output_file) with the
  # front end built by front_end_target, passing the options in ARGN.
  #
  # Both front ends run in the same working directory, which is part of the
  # module id generated for a source file that defines no external entities.
  get_filename_component(source_name "${source_file}" NAME)
  add_custom_command(
    OUTPUT "${output_file}"
    COMMAND "$<TARGET_FILE:${front_end_target}>" ${ARGN}
            "--gen_c_file_name=${output_file}" "${source_file}"
    DEPENDS ${front_end_target} "${source_file}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    COMMENT "Translating ${source_name} to C with ${front_end_target}"
    VERBATIM
  )
endfunction()

set(cpfe_c_files "")
set(runtime_c_files "")
set(cpfe_self_c_files "")
set(gen_c_file_names "")
foreach(source_file IN LISTS CORE_FRONT_END_SOURCE_FILES runtime_source_files)
  get_filename_component(name "${source_file}" NAME_WE)
  if(source_file IN_LIST runtime_source_files)
    # Some runtime source names (e.g., error.c) are also front end source
    # names.
    set(gen_c_file_name "rt_${name}.int.c")
    set(options ${runtime_options})
    list(APPEND runtime_c_files "${cpfe_c_dir}/${gen_c_file_name}")
  else()
    set(gen_c_file_name "${name}.int.c")
    set(options ${front_end_options})
    list(APPEND cpfe_c_files "${cpfe_c_dir}/${gen_c_file_name}")
  endif()
  list(APPEND gen_c_file_names "${gen_c_file_name}")
  add_translation(cpfe "${source_file}" "${cpfe_c_dir}/${gen_c_file_name}"
                  ${options})
  add_translation(cpfe-self "${source_file}"
                  "${cpfe_self_c_dir}/${gen_c_file_name}" ${options})
  list(APPEND cpfe_self_c_files "${cpfe_self_c_dir}/${gen_c_file_name}")
endforeach()

# Build cpfe-self from the generated C with only the options that C needs (not
# the C++ warning and code generation options inherited from the parent
# directory).
set_directory_properties(PROPERTIES
                         COMPILE_DEFINITIONS ""
                         COMPILE_OPTIONS ""
                         INCLUDE_DIRECTORIES ""
                         LINK_OPTIONS "")
if(MSVC)
  # The generated C is not written to be warning-free, and the objects for
  # the larger source files exceed the default limit on sections.
  add_compile_options(/w /bigobj)
else()
  # These match the options eccp uses to compile generated C (see the
  # edg_eccp_config files in bases/).
  add_compile_options(-w -falign-functions=4)
endif()
if($CACHE{32BIT_BUILD})
  add_compile_options(-m32 -msse2)
  add_link_options(-m32 -msse2)
endif()
add_library(cpfe-self-runtime STATIC ${runtime_c_files})
add_executable(cpfe-self ${cpfe_c_files})
# This uses the plain signature, as add_softfloat_to_target does.
target_link_libraries(cpfe-self cpfe-self-runtime)
if(MSVC)
  # For GCC and Clang, the template instantiations that each translation unit
  # gets are marked weak, but MSVC has no way to do that in C, so the same
  # instantiation is defined in several objects.  The definitions are
  # identical, so have the linker keep any one of them (as it would for a
  # COMDAT) instead of diagnosing them.
  target_link_options(cpfe-self PRIVATE /FORCE:MULTIPLE /IGNORE:4006,4088)
else()
  # The math library is linked implicitly only for C++.
  target_link_libraries(cpfe-self m)
endif()
add_softfloat_to_target(cpfe-self)

add_custom_target(cpfe-self-check
  COMMAND "${Python3_EXECUTABLE}"
          "${CMAKE_SOURCE_DIR}/dev_tools/bin/edg-compare-gen-c"
          "${cpfe_c_dir}" "${cpfe_self_c_dir}" ${gen_c_file_names}
  DEPENDS ${cpfe_self_c_files}
  COMMENT "Comparing the C generated by cpfe and cpfe-self"
  VERBATIM
)
